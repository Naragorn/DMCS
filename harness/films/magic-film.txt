# Film of the adrenaline magic (not part of the suite: tools/harness.sh harness/films/magic-film.txt). Three rounds,
# each with a fresh pair of NPCs (the target 2 m ahead, a neighbour beside it) and a full meter from two hits, then one
# spell through the game's own spell code (BasHarness player-cast): a fireball, a lightning bolt, a gravity push. Views:
# beside the NPCs and the player's own eyes. Sets only options an older build has too (no magic option), so the same
# scenario films the "before": there the spells are ordinary hits.
wait-level
frame-rate 90
press DMCS "Reset To Defaults"
set-option DMCS "Debug Logging" true
set-option DMCS "Adrenaline Per Hit" 50
set-option DMCS "Adrenaline Window" 20
set-option DMCS "Kneel And Execution" false
set-option DMCS "Charge Stagger Enabled" false
set-option DMCS "Clash of Might Mode" Off
player-invincible true

spawn-npc HumanMale distance=2.4 angle=28 brain=HumanDummy
spawn-npc HumanMale distance=2 angle=0 brain=HumanDummy
hit-npc Torso velocity=2
hit-npc Torso velocity=2
record-start fire fps=60 width=640 height=360 views=npc,first
wait 0.5
player-cast Fire
wait 4
record-stop
wait 3

spawn-npc HumanMale distance=2.4 angle=-28 brain=HumanDummy
spawn-npc HumanMale distance=2 angle=0 brain=HumanDummy
hit-npc Torso velocity=2
hit-npc Torso velocity=2
record-start lightning fps=60 width=640 height=360 views=npc,first
wait 0.5
player-cast Lightning
wait 4
record-stop
wait 3

spawn-npc HumanMale distance=2.4 angle=28 brain=HumanDummy
spawn-npc HumanMale distance=1.5 angle=0 brain=HumanDummy
hit-npc Torso velocity=2
hit-npc Torso velocity=2
record-start gravity fps=60 width=640 height=360 views=npc,first
wait 0.5
player-cast Gravity
wait 4
record-stop
quit
