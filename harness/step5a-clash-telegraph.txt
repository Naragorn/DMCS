# Step 5a: Clash of Might telegraph. A fighting NPC with a sword attacks the standing player: its big swings (long
# wind-up) flash at the middle of the blade, plain swings do not; with no blade in the player's hand the slow-motion
# is refused (BladeTooFar). The slow-motion itself (player blade near the flash, right angle to the swing) is
# unit-tested only: the harness cannot hold a blade against a real swing. Expects the defaults at the start.

wait-level
press DMCS "Reset To Defaults"
option-exists DMCS "Clash Only Swords" category="Clash of Might" type=Boolean interaction=ArrowList save=true
option-exists DMCS "Clash Min Wind Up" category="Clash of Might" type=Single interaction=Slider save=true
option-exists DMCS "Clash Slow Motion" category="Clash of Might" type=Boolean interaction=ArrowList save=true
option-exists DMCS "Clash Slow Motion Scale" category="Clash of Might" type=Single interaction=Slider save=true
option-exists DMCS "Clash Enemy Reach" category="Clash of Might" type=Single interaction=Slider save=true
option-exists DMCS "Clash Block Reach" category="Clash of Might" type=Single interaction=Slider save=true
option-exists DMCS "Clash Block Angle Tolerance" category="Clash of Might" type=Single interaction=Slider save=true
option-exists DMCS "Clash Flash Sound" category="Clash of Might" type=String interaction=ArrowList save=true
option-exists DMCS "Clash Flash Game Sound" category="Clash of Might" type=String interaction=ArrowList save=true
option-exists DMCS "Clash Flash Sound Volume" category="Clash of Might" type=Single interaction=Slider save=true
expect-option DMCS "Clash Flash Sound" "DMCS 1"
expect-option DMCS "Clash Flash Game Sound" Off
expect-option DMCS "Clash Flash Sound Volume" 0.8
option-exists DMCS "Clash Flash At Swing Top" category="Clash of Might" type=Boolean interaction=ArrowList save=true
expect-option DMCS "Clash Flash At Swing Top" true
expect-option DMCS "Clash Only Swords" true
expect-option DMCS "Clash Min Wind Up" 0.4
expect-option DMCS "Clash Slow Motion" true
expect-option DMCS "Clash Slow Motion Scale" 0.3
expect-option DMCS "Clash Enemy Reach" 2.5
expect-option DMCS "Clash Block Reach" 0.6
expect-option DMCS "Clash Block Angle Tolerance" 90
set-option DMCS "Debug Logging" true
# The NPC's hits on the player would fill the meter and its critical/kneel would end the fight
set-option DMCS "Adrenaline Enabled" false
set-option DMCS "Charge Stagger Enabled" false
# Disarm Strike stays on: the NPC's own swings into the standing player's motionless hand once counted as punches
# and disarmed it (2026-11-17); now they must not (Body Hit Min Hand Speed).
# A real NPC fights the player for two minutes: without this the player dies after about a minute (seen 2026-11-17)
player-invincible true

# 1. Big swings flash, plain swings do not; no blade in hand → no slow-motion
# The default loadout (CivilianMelee) is random and may add a shield, mace or axe; CivilianMeleeSword is only a sword,
# which the NPC draws from its back (an extra npc-grab gave it a second sword: two blades on the pictures, 2026-11-17).
spawn-npc HumanMale distance=1.5 angle=0 container=CivilianMeleeSword
mark
wait 60
expect-log [DMCS][Clash] HumanMale
expect-log flashes at the swing top (
expect-log s), sound DMCS 1 (3.60 s)
expect-log no slow-motion for
expect-log BladeTooFar
expect-no-log [DMCS][Disarm Strike] disarmed
expect-no-log slow-motion 0.30x
screenshot clash-npc of=npc

# 1b. A DMCS sound together with a game sound (any combination)
set-option DMCS "Clash Flash Game Sound" "Metal Hit"
mark
wait 20
expect-log s), sound DMCS 1 (3.60 s), game sound Metal Hit
set-option DMCS "Clash Flash Sound" Off
set-option DMCS "Clash Flash Game Sound" "Focus Ready"
mark
wait 20
expect-log s), game sound Focus Ready
expect-no-log sound DMCS
set-option DMCS "Clash Flash Game Sound" Off
# DMCS 2-4 (2026-11-20): each plays at the next flash, with its own length
set-option DMCS "Clash Flash Sound" "DMCS 2"
mark
await-log 30 s), sound DMCS 2 (1.60 s)
set-option DMCS "Clash Flash Sound" "DMCS 3"
mark
await-log 30 s), sound DMCS 3 (4.50 s)
set-option DMCS "Clash Flash Sound" "DMCS 4"
mark
await-log 30 s), sound DMCS 4 (3.00 s)
set-option DMCS "Clash Flash Sound" "DMCS 1"

# 1c. Flash at the swing top off: it shows as the wind-up starts
set-option DMCS "Clash Flash At Swing Top" false
mark
await-log 30 flashes as the wind-up starts (0.00 of
set-option DMCS "Clash Flash At Swing Top" true

# 2. Min wind-up above every attack of the game (max 0.77 s): nothing flashes any more
set-option DMCS "Clash Min Wind Up" 0.8
mark
wait 20
expect-log : not a big swing
expect-no-log : flash

# 3. Feature off: no clash lines at all
set-option DMCS "Clash Min Wind Up" 0.4
set-option DMCS "Clash of Might Mode" Off
mark
wait 15
expect-no-log [DMCS][Clash] HumanMale
quit
