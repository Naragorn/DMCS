# Step 1: the mod loads, the Mod Menu has every group with its Enabled toggle (Clash of Might: its mode),
# options apply live, Debug Logging gates the debug lines, Reset To Defaults works.

wait-level

# Loads
expect-log [DMCS] v0.4.0-beta.1 loaded from folder 'DMCS' with 159 mod options
expect-log [DMCS] Settings: Debug Logging=off | Clash of Might=Enabled | Disarm Strike=Enabled | Charge Stagger=Enabled | Adrenaline=Enabled | Parry=Enabled
expect-log [ModManager][ThunderScript][DMCS] Loaded ThunderScript: DMCS.DmcsScript

# Menu: groups, "Enabled" label (game's own localized text), types, tooltips
# Clash of Might has a mode dropdown instead of a toggle (user 2026-11-18: Off / Simple / On)
option-exists DMCS "Clash of Might Mode" category="Clash of Might" type=String save=true interaction=ArrowList tooltip="Simple"
option-exists DMCS "Disarm Strike Enabled" category="Disarm Strike" display=Enabled type=Boolean save=true interaction=ArrowList tooltip="staggered"
option-exists DMCS "Charge Stagger Enabled" category="Charge Stagger" display=Enabled type=Boolean save=true interaction=ArrowList tooltip="sprinting"
option-exists DMCS "Adrenaline Enabled" category=Adrenaline display=Enabled type=Boolean save=true interaction=ArrowList tooltip="critical"
option-exists DMCS "Parry Enabled" category=Parry display=Enabled type=Boolean save=true interaction=ArrowList tooltip="Counter Parry"
option-exists DMCS "Debug Logging" category=General display="Debug Logging" type=Boolean save=true interaction=ArrowList tooltip="Logs"
option-exists DMCS "Reset To Defaults" category=General display="Reset To Defaults" save=false interaction=ButtonList tooltip="default"

# Defaults
expect-option DMCS "Debug Logging" false
expect-option DMCS "Clash of Might Mode" Simple
expect-option DMCS "Disarm Strike Enabled" true
expect-option DMCS "Charge Stagger Enabled" true
expect-option DMCS "Adrenaline Enabled" true
expect-option DMCS "Parry Enabled" true
option-exists DMCS "Parry Needs Skill" category="Skill Tree" type=Boolean save=true interaction=ArrowList tooltip="Body tree"
expect-option DMCS "Parry Needs Skill" false

# Debug Logging on, then a feature toggle is logged live
mark
set-option DMCS "Debug Logging" true
expect-log [DMCS] Debug Logging on
expect-option DMCS "Debug Logging" true
mark
set-option DMCS "Clash of Might Mode" Off
expect-log [DMCS][Clash of Might] Disabled
expect-option DMCS "Clash of Might Mode" Off

# Debug Logging off: the toggle still switches, but no debug line
set-option DMCS "Debug Logging" false
mark
set-option DMCS "Clash of Might Mode" Simple
expect-option DMCS "Clash of Might Mode" Simple
expect-no-log [DMCS][Clash of Might] Enabled

# Reset To Defaults
set-option DMCS "Disarm Strike Enabled" false
set-option DMCS "Clash of Might Mode" On
set-option DMCS "Debug Logging" true
mark
press DMCS "Reset To Defaults"
expect-log [DMCS] Reset 158 options to defaults.
expect-option DMCS "Debug Logging" false
expect-option DMCS "Disarm Strike Enabled" true
expect-option DMCS "Clash of Might Mode" Simple
save-options

screenshot step1-arena
quit
