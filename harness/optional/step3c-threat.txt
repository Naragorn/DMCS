# Step 3c (optional, needs the Threat mod): DMCS finds Threat's surrender state by reflection.
# Run: DMCS_EXTRA_MODS=<path to HJ_Threat> tools/harness.sh harness/optional/step3c-threat.txt
# Threatening an NPC needs a blade at its neck and Threat's skill, which the harness cannot do yet:
# this checks the link only, the 100 % drop of a surrendered NPC is unit-tested.

wait-level
set-option DMCS "Debug Logging" true
spawn-npc HumanMale distance=4 angle=90 brain=HumanDummy
npc-grab DaggerCommon hand=right
mark
hit-npc Torso push=1
hit-npc Torso
expect-log Threat mod found
expect-no-log Threat mod not loaded
expect-no-log Exception
quit
