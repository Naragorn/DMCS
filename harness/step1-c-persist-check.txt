# Persistence part 2 (new game start, same harness prefix): the values from part 1 are back.
# The game applies saved values after the scripts load (Options.ApplyModOptions), so the check is the
# Settings line DMCS logs once the level has loaded.
wait-level
expect-log [DMCS] Settings: Debug Logging=on | Clash of Might=Enabled | Disarm Strike=Disabled | Charge Stagger=Enabled | Adrenaline=Enabled
expect-option DMCS "Debug Logging" true
expect-option DMCS "Disarm Strike Enabled" false
expect-option DMCS "Clash of Might Enabled" true
expect-option DMCS "Charge Speed Threshold" 6
expect-option DMCS "Charge Stagger Strength" "Knock Down"
expect-option DMCS "Charge Stagger Preset" Custom
# the rest of Realistic is back too: the preset row is not applied on load, so it cannot reset them
expect-option DMCS "Charge Stagger Chance" 75
expect-option DMCS "Charge Speed Window" 0.15
# Leave the prefix at defaults for the next run.
press DMCS "Reset To Defaults"
expect-option DMCS "Charge Speed Threshold" 4
expect-option DMCS "Charge Stagger Strength" Full
expect-option DMCS "Charge Stagger Preset" Gamey
save-options
quit
