# Step 3: Disarm Strike. A player hit on the hand of a staggered NPC makes it drop its weapon(s); it then reacts
# (by dice since 2026-11-22: a shock, backing off or cowering) and cannot pick up or draw a weapon meanwhile. The
# sections up to 6 fix the dice on the shock (passive HumanDummy brain) for 4 s; section 7 covers the other two.
# "Staggered" counts as it was before the hit: the stagger a hit causes does not let that same hit disarm.
# Expects the defaults at the start (runs after step2, which resets).

wait-level
# Start from the defaults even if an earlier scenario failed before its own reset (saved options carry over)
press DMCS "Reset To Defaults"
option-exists DMCS "Disarm Requires Stagger" category="Disarm Strike" type=Boolean save=true
option-exists DMCS "Disarm Include Forearm" category="Disarm Strike" type=Boolean save=true
option-exists DMCS "Disarm Drops Both Hands" category="Disarm Strike" type=Boolean save=true
option-exists DMCS "Disarm Chance" category="Disarm Strike" type=Int32 interaction=Slider save=true
option-exists DMCS "Disarm Stays Permanent" category="Disarm Strike" type=Boolean save=true
option-exists DMCS "Disarmed Shock Chance" category="Disarm Strike" type=Int32 interaction=Slider save=true
option-exists DMCS "Disarmed Shock Time" category="Disarm Strike" type=Single interaction=Slider save=true
option-exists DMCS "Disarmed Back Off Chance" category="Disarm Strike" type=Int32 interaction=Slider save=true
option-exists DMCS "Disarmed Back Off Time" category="Disarm Strike" type=Single interaction=Slider save=true
option-exists DMCS "Disarmed Cower Chance" category="Disarm Strike" type=Int32 interaction=Slider save=true
option-exists DMCS "Disarmed Cower Time" category="Disarm Strike" type=Single interaction=Slider save=true
expect-option DMCS "Disarm Requires Stagger" true
expect-option DMCS "Disarm Include Forearm" false
expect-option DMCS "Disarm Drops Both Hands" true
expect-option DMCS "Disarm Chance" 100
expect-option DMCS "Disarm Stays Permanent" false
expect-option DMCS "Disarmed Shock Chance" 50
expect-option DMCS "Disarmed Shock Time" 2
expect-option DMCS "Disarmed Back Off Chance" 50
expect-option DMCS "Disarmed Back Off Time" 2
expect-option DMCS "Disarmed Cower Chance" 50
expect-option DMCS "Disarmed Cower Time" 5
set-option DMCS "Debug Logging" true
# Adrenaline (step 4) would fill its meter from these hits and kill the NPC with a critical; tested in step4*.
set-option DMCS "Adrenaline Enabled" false
set-option DMCS "Disarmed Shock Chance" 100
set-option DMCS "Disarmed Back Off Chance" 0
set-option DMCS "Disarmed Cower Chance" 0
set-option DMCS "Disarmed Shock Time" 4
# This scenario checks the hand rule; body hits (Body Hit Drop, random) are covered by step3b.
set-option DMCS "Body Hit Drop" false

# A fighting NPC (its own brain must be swapped and given back), holding a weapon in each hand.
spawn-npc HumanMale distance=4 angle=90
npc-grab DaggerCommon hand=right
npc-grab DaggerCommon hand=left
expect-npc holding true

# 1. Not staggered: a hand hit does not disarm
mark
hit-npc RightHand
expect-npc holding true within=0
expect-log NotStaggered

# 2. Forearm is off by default
hit-npc Torso push=1
expect-npc staggered true
mark
hit-npc RightArm
expect-npc holding true within=0
expect-no-log [DMCS][Disarm Strike] disarmed

# 3. Staggered + hand hit: both hands drop, passive brain, no regrab
wait 3
hit-npc Torso push=1
expect-npc staggered true
mark
hit-npc RightHand
expect-npc holding false within=0.5
expect-npc brain HumanDummy within=0
expect-log [DMCS][Disarm Strike] disarmed
expect-log freezes in shock for 4 s, then goes for a weapon
# a weapon put back into its hand is released again
npc-grab DaggerCommon hand=right
expect-npc holding false within=1
expect-log while disarmed: released
# after the 4 s it gets its own brain back and may re-arm
expect-log may re-arm
expect-npc brain HumanMedium within=0
npc-grab DaggerCommon hand=right
expect-npc holding true within=1

# 4. Only the hit hand, with the setting off. The left dagger is given right before the hits (it replaces
#    whatever the hand holds): during a wait the fighting brain may put its left-hand item away for a rock
#    throw (BrainModuleThrow grabs rocks with the left hand) or pick up a shield lying around, and an NPC
#    holding a buckler did not stagger from the push=1 test hit (seen 2026-11-16) - none of it from DMCS.
set-option DMCS "Disarm Drops Both Hands" false
wait 3
npc-grab DaggerCommon hand=left
hit-npc Torso push=1
expect-npc staggered true
mark
hit-npc LeftHand
expect-log disarmed
expect-npc holding true within=0
set-option DMCS "Disarm Drops Both Hands" true

# 5. Charge stagger then disarm (the intended combo); the charge hit itself does not disarm
expect-log may re-arm
npc-grab DaggerCommon hand=left
expect-npc holding true
run 1
wait 1.5
mark
hit-npc RightHand
expect-npc stagger Full within=0
expect-npc holding true within=0
hit-npc RightHand
expect-npc holding false within=0.5
expect-log disarmed
stop-run

# 6. Disabled: no disarm, live
expect-log may re-arm
set-option DMCS "Disarm Strike Enabled" false
npc-grab DaggerCommon hand=right
wait 3
hit-npc Torso push=1
expect-npc staggered true
mark
hit-npc RightHand
expect-npc holding true within=0
expect-no-log [DMCS][Disarm Strike] disarmed
set-option DMCS "Disarm Strike Enabled" true

# 7. The other two reactions (user 2026-11-22): backing off walks the NPC away from the player with its own brain's
#    move module (its tree paused), cowering is the game's own BrainModuleFear panic. Both end with the own brain.
#    Not run yet (built on Windows); how they look: headset.
expect-log may re-arm
set-option DMCS "Disarmed Shock Chance" 0
set-option DMCS "Disarmed Back Off Chance" 100
npc-grab DaggerCommon hand=right
wait 3
hit-npc Torso push=1
expect-npc staggered true
mark
hit-npc RightHand
expect-log backs off for 2 s, then goes for a weapon
await-log 4 is done with its reaction (backs off)
set-option DMCS "Disarmed Back Off Chance" 0
set-option DMCS "Disarmed Cower Chance" 100
npc-grab DaggerCommon hand=right
wait 3
hit-npc Torso push=1
expect-npc staggered true
mark
hit-npc RightHand
expect-log cowers for 5 s, then goes for a weapon
await-log 7 is done with its reaction (cowers)

press DMCS "Reset To Defaults"
expect-no-log Exception
screenshot step3-disarm-strike
quit
